cmake_minimum_required(VERSION 3.16)
project(phantom_tunnel_jni CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT ANDROID)
  find_package(JNI REQUIRED COMPONENTS JVM)
endif()
if(NOT PHANTOM_TUNNEL_LIBRARY)
  message(FATAL_ERROR "Set PHANTOM_TUNNEL_LIBRARY to the Rust cdylib for this target ABI")
endif()
add_library(phantom_tunnel_native SHARED IMPORTED)
set_target_properties(phantom_tunnel_native PROPERTIES IMPORTED_LOCATION "${PHANTOM_TUNNEL_LIBRARY}")
if(WIN32)
  if(NOT PHANTOM_TUNNEL_IMPLIB)
    message(FATAL_ERROR "Set PHANTOM_TUNNEL_IMPLIB to the Rust .dll.lib import library")
  endif()
  set_target_properties(phantom_tunnel_native PROPERTIES IMPORTED_IMPLIB "${PHANTOM_TUNNEL_IMPLIB}")
else()
  set_target_properties(phantom_tunnel_native PROPERTIES IMPORTED_NO_SONAME TRUE)
endif()
add_library(phantom_tunnel_jni SHARED bridge.cpp)
target_include_directories(phantom_tunnel_jni PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(phantom_tunnel_jni PRIVATE phantom_tunnel_native)
get_filename_component(PHANTOM_LIBRARY_DIRECTORY "${PHANTOM_TUNNEL_LIBRARY}" DIRECTORY)
if(APPLE)
  set_target_properties(phantom_tunnel_jni PROPERTIES BUILD_RPATH "${PHANTOM_LIBRARY_DIRECTORY};@loader_path")
else()
  set_target_properties(phantom_tunnel_jni PROPERTIES BUILD_RPATH "${PHANTOM_LIBRARY_DIRECTORY};$ORIGIN")
endif()
