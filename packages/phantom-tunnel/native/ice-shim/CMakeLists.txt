cmake_minimum_required(VERSION 3.16)
project(phantom_ice C)
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(NO_SERVER ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBJUICE_SOURCE}" libjuice)
add_library(phantom_ice STATIC shim.c)
target_compile_definitions(phantom_ice PRIVATE JUICE_STATIC)
target_link_libraries(phantom_ice PRIVATE juice-static)
set_target_properties(phantom_ice PROPERTIES POSITION_INDEPENDENT_CODE ON)
install(TARGETS phantom_ice juice-static ARCHIVE DESTINATION lib)
